from typing import List, Dict, Tuple

class Tile:
    def __init__(self, x: int, y: int, region: str, biome: str, danger_level: int):
        self.x = x
        self.y = y
        self.region = region
        self.biome = biome
        self.danger_level = danger_level
        self.description = ""
        self.safezone = False
        self.npcs: List[str] = []
        self.services: List[str] = []
        self.enemies: List[str] = []

    def to_dict(self):
        return {
            "x": self.x,
            "y": self.y,
            "region": self.region,
            "biome": self.biome,
            "danger_level": self.danger_level,
            "description": self.description,
            "safezone": self.safezone,
            "npcs": self.npcs,
            "services": self.services,
            "enemies": self.enemies
        }

class World:
    def __init__(self, width: int = 300, height: int = 300):
        self.width = width
        self.height = height
        self.world_map: Dict[Tuple[int, int], Tile] = {}
        self._initialize_static_world()

    def add_tile(self, x: int, y: int, region: str, biome: str, danger_level: int,
                 description: str = "", safezone: bool = False,
                 npcs: List[str] = None, services: List[str] = None,
                 enemies: List[str] = None):
        tile = Tile(x, y, region, biome, danger_level)
        tile.description = description
        tile.safezone = safezone
        tile.npcs = npcs or []
        tile.services = services or []
        tile.enemies = enemies or []
        self.world_map[(x, y)] = tile

    def get_tile(self, x: int, y: int) -> Tile:
        return self.world_map.get((x, y))
